#include <algorithm> #include <stdexcept> #include <vector> class MinHeap { std::vector<int> heap; void percDown(int i) { while (2 * i + 1 < static_cast<int>(heap.size())) { int child = 2 * i + 1; if (child + 1 < static_cast<int>(heap.size()) && heap[child + 1] < heap[child]) ++child; if (heap[i] <= heap[child]) break; std::swap(heap[i], heap[child]); i = child; } } public: int delMin() { if (heap.empty()) throw std::underflow_error("empty heap"); std::swap(heap[0], heap.back()); int m = heap.back(); heap.pop_back(); if (!heap.empty()) percDown(0); return m; } };
